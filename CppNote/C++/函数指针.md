使用
```cpp
#include <iostream>
static void printHelloWorld() {
	std::cout << "Hello, World!" << std::endl;
}
static void printA(int a) {
	std::cout << a << std::endl;
}
int main() {
	auto function = printHelloWorld;// 函数指针，&可以省略，也就是说函数名就是函数的地址
	function();

	auto function2 = printA;
	function2(10);

	typedef void (*FunctionType)(); // 使用typedef定义函数指针类型，相比auto，typedef可以让代码更清晰
	FunctionType function3 = printHelloWorld;
	function3();

	typedef void (*FunctionType2)(int);
	FunctionType2 function4 = printA;
	function4(20);
	return 0;
}
```

应用
```cpp
#include <iostream>
#include <vector>
static void printValue(int value) {
	std::cout << "Value: " << value << std::endl;
}
static void ForEach(const std::vector<int>& vec, void (*func)(int)) {
	for (int value : vec) {
		func(value);
	}
}
int main() {
	std::vector<int> vec = { 1, 2, 3, 4, 5 };
	ForEach(vec, printValue);
	return 0;
}
```

扩展[[lambda]]
```cpp
#include <iostream>
#include <vector>
static void ForEach(const std::vector<int>& vec, void (*func)(int)) {
	for (int value : vec) {
		func(value);
	}
}
int main() {
	std::vector<int> vec = { 1, 2, 3, 4, 5 };
	ForEach(vec, [](int value) {
		std::cout << "Value: " << value << std::endl;
		});
	return 0;
}
```
