运算符本质就是函数
```cpp
#include <iostream>
class Entity {
public:
	int a;
	double b;
	Entity() : a(0), b(0.0) {}
	Entity(int _a, double _b) : a(_a), b(_b) {}
	Entity operator+(const Entity& other) const {
		return Entity(a + other.a, b + other.b);
	}
	Entity Add(const Entity& other) const {
		return Entity(a + other.a, b + other.b);
		// 等价于以下两种方式调用 operator+ 函数
		//return *this + other;
		//return operator+(other);
	}
	bool operator==(const Entity& othrt) const {
		return a == othrt.a && b == othrt.b;
	}
	bool Equals(const Entity& other) const {
		return operator==(other);
	}
};
static std::ostream& operator<<(std::ostream& os, const Entity& e) {
	os << e.a << ", " << e.b;
	return os;
}
int main() {
	Entity e1(1, 2.0);
	Entity e2(3, 4.5);
	Entity e3 = e1 + e2; // 调用 operator+ 函数
	std::cout << e3 << std::endl; // 输出 4, 6
	std::cout << (e1 == e2) << std::endl;
	return 0;
}
```
