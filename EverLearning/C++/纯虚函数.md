纯虚函数是在基类定义一个没有实现的函数并强制子类实现该函数
```cpp
#include <iostream>
class Entity {
public:
	virtual std::string GetName() const = 0;
};
class Player : public Entity {
public:
	std::string GetName() const override { return "Player"; }
};
int main() {
	//Entity entity; // 不可以实例化抽象类
	Player player; // 可以实例化派生类，如果派生类没有实现纯虚函数，则也不能实例化
	return 0;
}
```
