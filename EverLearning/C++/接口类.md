接口类是特殊的[[抽象类]]，所有函数都是纯虚函数、没有成员变量、没有具体实现的类
例子：
```cpp
#include <iostream>
class Printable {
public:
	virtual std::string GetClassName() const = 0; // 纯虚函数
};
static void Print(const Printable& obj) {
	std::cout << obj.GetClassName() << std::endl;
}
class Entity : public Printable {
public:
	std::string GetClassName() const override {
		return "Entity";
	}
};
class Player : public Printable {
public:
	std::string GetClassName() const override {
		return "Player";
	}
};
int main() {
	Entity entity;
	Print(entity);
	Player player;
	Print(player);
	return 0;
}
```
