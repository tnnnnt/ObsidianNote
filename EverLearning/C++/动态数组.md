```cpp
#include <iostream>
#include <vector>
class Entity {
public:
	int x, y;
	Entity(int x, int y) : x(x), y(y) {}
	Entity(const Entity& other) : x(other.x), y(other.y) {
		std::cout << "Copied!" << std::endl;
	}
};
int main() {
	// 优化前，6次copy
	std::vector<Entity> entities;
	entities.push_back(Entity(1, 2));
	entities.push_back(Entity(3, 4));
	entities.push_back(Entity(5, 6));
}
```

```cpp
	std::vector<Entity> entities;
```

```cpp
#include <iostream>
#include <vector>
class Entity {
public:
	int x, y;
	Entity(int x, int y) : x(x), y(y) {}
	Entity(const Entity& other) : x(other.x), y(other.y) {
		std::cout << "Copied!" << std::endl;
	}
};
int main() {
	// push_back改为emplace_back，直接在容器中构造对象，变为0次copy
	std::vector<Entity> entities;
	entities.reserve(3);
	entities.emplace_back(1, 2);
	entities.emplace_back(3, 4);
	entities.emplace_back(5, 6);
}
```
