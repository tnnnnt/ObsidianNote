```cpp
#include <iostream>
class Entity {
public:
	int X, Y;
	void Move(int x, int y) {
		X += x;
		Y += y;
	}
};
class Player : public Entity {
public:
	const char* Name;
	void Print() {
		std::cout << Name << " is at (" << X << ", " << Y << ")" << std::endl;
	}
};
int main() {
	Player player{};
	player.X = 2;
	player.Y = 3;
	player.Name = "John";
	player.Move(5, 7);
	player.Print();
	return 0;
}
```
