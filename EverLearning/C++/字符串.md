前置知识：[[指针]]、[[数组]]
```cpp
#include <iostream>
#include <string>
int main() {
	// C语言风格的字符串
	const char* name = "tnt";//必须是const char*，因为字符串字面量是只读的
	char name2[] = "tnt";//隐藏了'\0'，所以是4字节
	char name3[] = { 't','n','t','\0' };//必须以'\0'结尾，否则会出现未定义行为
	std::cout << sizeof(name) << std::endl;//sizeof(name)是指针的大小，通常是8字节
	std::cout << sizeof(name2) << std::endl;//sizeof(name2)是数组的大小，隐藏了'\0'，所以是4字节
	std::cout << sizeof(name3) << std::endl;//sizeof(name3)是数组的大小，需要包含'\0'，所以是4字节
	// C++风格的字符串
	std::string name4 = "tnt";//std::string是一个类，内部封装了一个char数组和一些成员函数
	std::cout << sizeof(name4) << std::endl;//sizeof(name4)是类的大小，通常是32字节
	// string有很多成员函数，不需要刻意去记忆，查文档即可
}
```
