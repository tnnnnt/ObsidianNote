在每次销毁对象时运行的特殊函数
[[构造函数]]
```cpp
#include <iostream>
class Entity {
public:
	Entity() {
		std::cout << "Created Entity" << std::endl;
	}
	~Entity() {
		std::cout << "Destroyed Entity" << std::endl;
	}
	void Print() {
		std::cout << "print" << std::endl;
	}
};
int main() {
	std::cout << "Hello World!" << std::endl;
	{
		Entity e;
		e.Print();
	}
	std::cout << "Goodbye World!" << std::endl;
	return 0;
}
```
