#include <iostream>
#include <memory>
class Entity {
public:
	Entity() {
		std::cout << "Created Entity!" << std::endl;
	}
	~Entity() {
		std::cout << "Destroyed Entity!" << std::endl;
	}
};
int main() {
	{
		//std::unique_ptr<Entity> entity = new Entity(); // 错误，不能直接将裸指针赋值给unique_ptr
		//std::unique_ptr<Entity> entity(new Entity()); // 正确，直接使用裸指针创建unique_ptr，但不推荐这种方式
		std::unique_ptr<Entity> entity = std::make_unique<Entity>(); // 正确，使用make_unique创建unique_ptr
		//std::unique_ptr<Entity> e0 = entity; // 错误，unique_ptr不允许拷贝构造
		std::unique_ptr<Entity> e0 = std::move(entity); // 将entity的所有权转移给e0
	}
	{
		std::shared_ptr<Entity> e0;
		{
			std::shared_ptr<Entity> entity = std::make_shared<Entity>(); // 正确，使用make_shared创建shared_ptr
			e0 = entity; // 正确，shared_ptr允许拷贝构造，引用计数会增加
		}
	}
	{
		std::weak_ptr<Entity> e0;
		{
			std::shared_ptr<Entity> entity = std::make_shared<Entity>(); // 正确，使用make_shared创建shared_ptr
			e0 = entity; // 正确，weak_ptr允许拷贝构造，但是不会增加引用计数
		}
	}
}
