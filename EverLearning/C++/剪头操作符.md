```cpp
#include <iostream>
class Entity {
public:
	int x;
	void Print() const {
		std::cout << "tnt" << std::endl;
	}
};
int main() {
	Entity e{};
	e.x = 1;
	e.Print();
	
	Entity* p = &e;
	p->x = 2; // 等价于(*p).x=2
	p->Print(); // 等价于(*p).Print()
}
```
[[运算符重载]]
```cpp
#include <iostream>
class Entity {
public:
	void Print() const {
		std::cout << "tnt" << std::endl;
	}
};
class ScopedPtr {
private:
	Entity* m_Obj;
public:
	ScopedPtr(Entity* e) :m_Obj(e) {
	}
	~ScopedPtr() {
		delete m_Obj;
	}
	const Entity* operator->() const {
		return m_Obj;
	}
	Entity* operator->() {
		return m_Obj;
	}
};
int main() {
	const ScopedPtr entity = new Entity();
	entity->Print();
	ScopedPtr e = new Entity();
	e->Print();
}
```
应用：获取偏移量
```cpp
#include <iostream>
struct Entity {
	int x, y;
	float z;
};
int main() {
	const int offsetX = (int)&((Entity*)nullptr)->x;
	const int offsetY = (int)&((Entity*)nullptr)->y;
	const int offsetZ = (int)&((Entity*)nullptr)->z;
	std::cout << offsetX << std::endl;
	std::cout << offsetY << std::endl;
	std::cout << offsetZ << std::endl;
}
```
