```cpp
#include <iostream>
class String {
private:
	char* str;
	unsigned int length;
public:
	String(const char* s) {
		length = strlen(s);
		str = new char[length + 1];
		memcpy(str, s, length);
		str[length] = 0;
	}
	~String() {
		delete[] str;
	}
	friend std::ostream& operator<<(std::ostream& os, const String& s);
};
std::ostream& operator<<(std::ostream& os, const String& s) {
	os << s.str;
	return os;
}
int main() {
	String s = "tnt";
	String s2 = s; // 浅拷贝，s2.str指向s.str的内存地址，析构函数会释放s.str的内存，导致s2.str悬空指针
	std::cout << s << std::endl;
	std::cout << s2 << std::endl;
}
```

```cpp
#include <iostream>
class String {
private:
	char* str;
	unsigned int length;
public:
	String(const char* s) {
		length = strlen(s);
		str = new char[length + 1];
		memcpy(str, s, length);
		str[length] = 0;
	}
	~String() {
		delete[] str;
	}
	String(const String& s) :length(s.length) {
		str = new char[length + 1];
		memcpy(str, s.str, length + 1);
	}
	friend std::ostream& operator<<(std::ostream& os, const String& s);
};
std::ostream& operator<<(std::ostream& os, const String& s) {
	os << s.str;
	return os;
}
int main() {
	String s = "tnt";
	String s2 = s; // 深拷贝，s和s2指向不同的内存空间
	std::cout << s << std::endl;
	std::cout << s2 << std::endl;
}
```
