```cpp
#include <iostream>
class Entity {
};
static Entity* CreateEntity() {
    return new Entity();
}
int main() {
    // 在栈上创建对象
    // 作用域结束后会自动销毁
    Entity e;
    // 在堆上创建对象
    Entity* e2 = CreateEntity();
    // 使用完堆上的对象后，手动释放内存
    delete e2;
    // delete 不会自动将指针设置为 nullptr
    // 所以需要手动设置
    e2 = nullptr;
    return 0;
}
```
[[new]]
使用[[智能指针]]就不需要手动[[delete]]了
