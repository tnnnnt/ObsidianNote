```cpp
int* CreateArray() {
	int array[50];
	return array;
}

int* CreateArray() {
	int* array = new int[50];
```

```cpp
#include <iostream>
class Entity {
public:
	Entity() {
		std::cout << "Created Entity!" << std::endl;
	}
	~Entity() {
		std::cout << "Destroyed Entity!" << std::endl;
	}
};
class ScopedPtr {
private:
	Entity* m_Obj;
public:
	ScopedPtr(Entity* obj) : m_Obj(obj) {}
	~ScopedPtr() {
		delete m_Obj;
	}
	Entity* operator->() {
		return m_Obj;
	}
};
int main() {
	{
		ScopedPtr entity(new Entity());
	}
}
```
