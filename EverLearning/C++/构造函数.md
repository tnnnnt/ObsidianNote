在每次实例化对象时运行的特殊函数
```cpp
#include <iostream>
class Entity {
public:
	int X;
	int Y;
	// 无参构造函数
	Entity() {
		X = 0;
		Y = 0;
	}
	// 有参构造函数
	Entity(int x, int y) {
		X = x;
		Y = y;
	}
	void Print() const {
		std::cout << X << ", " << Y << std::endl;
	}
};
int main() {
	Entity e1; // 调用无参构造函数
	Entity e2(5, 10); // 调用有参构造函数
	e1.Print(); // 输出: 0, 0
	e2.Print(); // 输出: 5, 10
	return 0;
}
```
技巧：不希望用户创建实例，可以把构造函数设置为private
```cpp
#include <iostream>
class Log {
private:
	Log() {} // 私有构造函数，防止外部创建对象
public:
	static void WriteLog(const std::string& message) {
		std::cout << "Log: " << message << std::endl;
	}
};
int main() {
	// Log log; // 不能创建对象，因为构造函数是私有的
	Log::WriteLog("This is a log message.");
	return 0;
}
```
注意：不写构造函数也会有隐藏的默认无参构造函数
