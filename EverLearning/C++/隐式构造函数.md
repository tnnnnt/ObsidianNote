```cpp
#include <iostream>
class Entity {
private:
	int a;
public:
	Entity() : a(0) {}
	Entity(int value) : a(value) {}
};
class Entity2 {
private:
	int a;
public:
	Entity2() : a(0) {}
	explicit Entity2(int value) : a(value) {}
};
int main() {
	Entity e1 = 10; // 允许隐式转换
	//Entity2 e2 = 10; // 错误，explicit阻止了隐式转换
	Entity2 e2(10); // 正确，显式调用构造函数
	return 0;
}
```
在构造函数前加关键字"explicit"可以阻止隐式转换
